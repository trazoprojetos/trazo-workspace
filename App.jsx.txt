const convertFileToBase64 = (file) => {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.readAsDataURL(file);
    reader.onload = () => resolve(reader.result);
    reader.onerror = (error) => reject(error);
  });
};

          </nav>
        </div>

        {/* Login Switcher Area */}
        <div className="p-4 m-4 rounded-2xl bg-[#080d1e] border border-slate-800 text-xs text-slate-300 space-y-3 shadow-xl">
          <div className="flex items-center justify-between font-bold text-slate-200 border-b border-slate-800 pb-2">
            <div className="flex items-center space-x-2">
              <MemberAvatar name={activeMember.name} avatarUrl={activeMember.avatarUrl} size="w-8 h-8" />
              <div className="truncate max-w-[100px]">
                <div className="font-bold text-slate-100 text-xs truncate">{activeMember.name}</div>
                <div className="text-[9px] text-slate-400 truncate">{activeMember.role}</div>
              </div>
            </div>
            <span className={`text-[9px] px-2 py-0.5 rounded font-mono font-bold ${
              activeMember.isSystemAdmin ? 'bg-amber-500/20 text-amber-300 border border-amber-500/30' : 'bg-cyan-500/20 text-cyan-300'
            }`}>
              {activeMember.isSystemAdmin ? 'MASTER' : 'MEMBRO'}
            </span>
          </div>

          <form onSubmit={handleLoginSubmit} className="space-y-2 pt-1">
            <label className="text-[10px] uppercase font-bold text-slate-400 block">Autenticar Conta (E-mail + PIN):</label>
            <input 
              type="email" 
              placeholder="E-mail do membro"
              value={loginEmail}
              onChange={e => setLoginEmail(e.target.value)}
              className="bg-slate-950 border border-slate-800 rounded-lg p-2 w-full text-xs text-slate-200 outline-none focus:border-cyan-500"
              required
            />
            <div className="flex space-x-1.5">
              <input 
                type="password" 
                placeholder="PIN"
                value={loginPin}
                onChange={e => setLoginPin(e.target.value)}
                className="bg-slate-950 border border-slate-800 rounded-lg p-2 w-20 text-xs text-slate-200 outline-none focus:border-cyan-500"
                required
              />
              <button type="submit" className="flex-1 bg-cyan-600 hover:bg-cyan-500 text-white font-bold rounded-lg text-xs transition-colors">
                Entrar
              </button>
            </div>
            {loginError && <p className="text-[10px] text-rose-400 font-medium">{loginError}</p>}
          </form>
        </div>
      </aside>

      {/* Main Container */}
      <main className="flex-1 flex flex-col h-screen overflow-hidden z-10 min-w-0">
        <header className="px-6 py-4 flex flex-col sm:flex-row justify-between items-start sm:items-center border-b border-slate-800/80 bg-[#0a1128]/60 backdrop-blur-md gap-4">
          <div>
            <h2 className="text-xl font-extrabold text-slate-100 flex items-center space-x-2">
              <span>
                {currentView === 'kanban' && 'Painel Kanban Operacional'}
                {currentView === 'responsibilities' && 'Painel de Responsabilidades'}
                {currentView === 'financial_admin' && 'Painel Financeiro Geral (Master)'}
                {currentView === 'financial_ops' && 'Painel Financeiro Operacional'}
                {currentView === 'clients' && 'Gestão de Clientes Cadastrados'}
                {currentView === 'members' && 'Equipe & Gestão de Permissões'}
              </span>
            </h2>
            <p className="text-xs text-slate-400">
              {!activeMember.isSystemAdmin && `Acesso limitado ao perfil operacional de: ${activeMember.name}`}
            </p>
          </div>

          <div className="flex flex-wrap items-center gap-2 w-full sm:w-auto">
            {currentView === 'kanban' && activeMember.isSystemAdmin && (
              <button
                onClick={() => { setEditingColumn(null); setIsColumnModalOpen(true); }}
                className="flex items-center space-x-1.5 bg-slate-800 hover:bg-slate-700 text-slate-200 px-3 py-2 rounded-xl text-xs font-bold border border-slate-700"
              >
                <Plus size={14} />
                <span>Nova Coluna</span>
              </button>
            )}

            <button
              onClick={() => {
                if (currentView === 'members') {
                  if (!activeMember.isSystemAdmin) {
                    showToast('Apenas o usuário Master pode cadastrar membros!', 'error');
                    return;
                  }
                  setEditingMember(null);
                  setIsMemberModalOpen(true);
                } else if (currentView === 'clients') {
                  setEditingClient(null);
                  setIsClientModalOpen(true);
                } else {
                  setEditingCard(null);
                  setActiveCardTab('overview');
                  setIsCardModalOpen(true);
                }
              }}
              className="flex items-center space-x-2 bg-cyan-600 hover:bg-cyan-500 text-white px-4 py-2 rounded-xl text-sm font-bold shadow-lg ml-auto sm:ml-0 transition-all"
            >
              <Plus size={16} />
              <span>
                {currentView === 'members' ? 'Novo Membro' : currentView === 'clients' ? 'Novo Cliente' : 'Novo Cartão'}
              </span>
            </button>
          </div>
        </header>

        {/* Dynamic View Loader */}
        <div className="flex-1 overflow-auto p-4 sm:p-6">
          {currentView === 'kanban' && (
            <KanbanBoard 
              cards={visibleCards} 
              clients={clients} 
              members={members}
              columns={columns}
              activeMember={activeMember}
              onCardClick={(card) => { setEditingCard(card); setActiveCardTab('overview'); setIsCardModalOpen(true); }}
              onUpdateStatus={updateCardStatus}
              onToggleTask={handleToggleTask}
              onEditColumn={(col) => { setEditingColumn(col); setIsColumnModalOpen(true); }}
              onDeleteColumn={handleDeleteColumn}
            />
          )}

          {currentView === 'responsibilities' && (
            <ResponsibilitiesView 
              cards={visibleCards} 
              clients={clients} 
              members={members}
              activeMember={activeMember}
              onCardClick={(card) => { setEditingCard(card); setActiveCardTab('overview'); setIsCardModalOpen(true); }}
              onToggleTask={handleToggleTask}
            />
          )}

          {currentView === 'financial_admin' && (activeMember.isSystemAdmin || activeMember.canAccessFinancialAdmin) && (
            <FinancialAdminDashboard 
              cards={cards} 
              clients={clients}
              onCardClick={(card) => { setEditingCard(card); setActiveCardTab('finance'); setIsCardModalOpen(true); }}
            />
          )}

          {currentView === 'financial_ops' && (activeMember.isSystemAdmin || activeMember.canAccessFinancialOps) && (
            <FinancialOpsDashboard 
              cards={cards} 
              clients={clients}
              onCardClick={(card) => { setEditingCard(card); setActiveCardTab('finance'); setIsCardModalOpen(true); }}
            />
          )}

          {currentView === 'clients' && (activeMember.isSystemAdmin || activeMember.canAccessClients) && (
            <ClientsView 
              clients={clients} 
              onEditClient={(client) => { setEditingClient(client); setIsClientModalOpen(true); }}
              onDeleteClient={handleDeleteClient}
            />
          )}

          {currentView === 'members' && (activeMember.isSystemAdmin || activeMember.canAccessMembers) && (
            <MembersView 
              members={members} 
              activeMember={activeMember}
              onEditMember={(member) => { setEditingMember(member); setIsMemberModalOpen(true); }}
              onDeleteMember={handleDeleteMember}
            />
          )}
        </div>
      </main>

      {/* --- MODALS --- */}
      {isColumnModalOpen && (
        <ColumnModal
          column={editingColumn}
          onClose={() => setIsColumnModalOpen(false)}
          onSave={handleSaveColumn}
        />
      )}

      {isMemberModalOpen && (
        <MemberModal 
          member={editingMember} 
          activeMember={activeMember}
          onClose={() => setIsMemberModalOpen(false)} 
          onSave={handleSaveMember} 
        />
      )}

      {isClientModalOpen && (
        <ClientModal 
          client={editingClient} 
          onClose={() => setIsClientModalOpen(false)} 
          onSave={handleSaveClient} 
        />
      )}

      {isCardModalOpen && (
        <CardModal 
          card={editingCard} 
          clients={clients} 
          members={members}
          columns={columns}
          onClose={() => setIsCardModalOpen(false)} 
          onSave={handleSaveCard} 
          onDelete={handleDeleteCard}
          activeTab={activeCardTab}
          setActiveTab={setActiveCardTab}
          setLightboxImage={setLightboxImage}
        />
      )}

      {lightboxImage && (
        <div 
          className="fixed inset-0 z-50 bg-slate-950/90 backdrop-blur-xl flex items-center justify-center p-4"
          onClick={() => setLightboxImage(null)}
        >
          <div className="relative max-w-4xl w-full max-h-[90vh] flex flex-col items-center">
            <button 
              onClick={() => setLightboxImage(null)}
              className="absolute -top-12 right-0 p-2 bg-slate-800 text-white rounded-full"
            >
              <X size={24} />
            </button>
            <img src={lightboxImage} alt="Preview" className="max-w-full max-h-[85vh] rounded-2xl object-contain shadow-2xl border border-white/10" />
          </div>
        </div>
      )}
    </div>
  );
}
